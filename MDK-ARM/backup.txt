/**************************************************************************
函数功能：按键扫描
入口参数：双击等待时间
返回  值：按键状态 0：无动作 1：单击 2：双击
**************************************************************************/
u8 click_N_Double (u8 time)
{
    static	u8 flag_key, count_key, double_key;
    static	u16 count_single, Forever_count;
    if(KEY == 0)  Forever_count++; //长按标志位未置1
    else        Forever_count = 0;
    if(0 == KEY && 0 == flag_key)		flag_key = 1;
    if(0 == count_key)
    {
        if(flag_key == 1)
        {
            double_key++;
            count_key = 1;
        }
        if(double_key == 2)
        {
            double_key = 0;
            count_single = 0;
            return 2;//双击执行的指令
        }
    }
    if(1 == KEY)			flag_key = 0, count_key = 0;

    if(1 == double_key)
    {
        count_single++;
        if(count_single > time && Forever_count < time)
        {
            double_key = 0;
            count_single = 0;
            return 1;//单击执行的指令
        }
        if(Forever_count > time)
        {
            double_key = 0;
            count_single = 0;
        }
    }
    return 0;
}

函数功能：按键扫描
入口参数：无

u8 click(void)
{
    static u8 flag_key = 1; //按键按松开标志
    if(flag_key && KEY == 0)
    {
        flag_key = 0;
        return 1;	// 按键按下
    }
    else if(1 == KEY)			flag_key = 1;
    return 0;//无按键按下
}

/**************************************************************************
函数功能：检测小车是否被拿起
入口参数：int
返回  值：unsigned int
**************************************************************************/
int Pick_Up(float Acceleration, float Angle, int encoder_left, int encoder_right)
{
    static u16 flag, count0, count1, count2;
    if(flag == 0)                                                                 //第一步
    {
        if(myabs(encoder_left) + myabs(encoder_right) < 700)                     //条件1，小车接近静止
            count0++;
        else
            count0 = 0;
        if(count0 > 10)
            flag = 1, count0 = 0;
    }
    if(flag == 1)                                                                //进入第二步
    {
        if(++count1 > 400)       count1 = 0, flag = 0;                          //超时不再等待2000ms
        if(Acceleration > 28000 && (Angle > (-20 + Zhongzhi)) && (Angle < (20 + Zhongzhi)) && Flag_Qian != 1 && Flag_Hou != 1 && Flag_Left != 1 && Flag_Right != 1) //条件2，小车是在0度附近被拿起
            flag = 2;
    }
    if(flag == 2)                                                                //第三步
    {
        if(++count2 > 200)       count2 = 0, flag = 0;                            //超时不再等待1000ms
        if(myabs(encoder_left + encoder_right) > 9900)                             //条件3，小车的轮胎因为正反馈达到最大的转速
        {
            flag = 0;
            return 1;                                                               //检测到小车被拿起
        }
    }
    return 0;
}

/**************************************************************************
函数功能：采集遥控器的信号
入口参数：无
返回  值：无
**************************************************************************/
void Get_Elrs()
{

    Remoter_Ch1 = map(CRSF_RX_packet.CH[1], CRSF_CHANNEL_VALUE_MIN, CRSF_CHANNEL_VALUE_MAX, -1000, 1000);
    Remoter_Ch2 =  map(CRSF_RX_packet.CH[3], CRSF_CHANNEL_VALUE_MIN, CRSF_CHANNEL_VALUE_MAX, -1000, 1000);
    Arm_ch6 = map(CRSF_RX_packet.CH[5], CRSF_CHANNEL_VALUE_MIN, CRSF_CHANNEL_VALUE_MAX, -1000, 1000);


    // printf( "ch1 %f, ch2 %f ,arm ch6 is %f \r\n", Remoter_Ch1,Remoter_Ch2,Arm_ch6);



    float a = atan2(Remoter_Ch1, Remoter_Ch2);
    float p = sqrt(pow(Remoter_Ch1, 2) + pow(Remoter_Ch2, 2));

    if (p > 200)
    {
        if (a > Math_PI / 4 && a <= Math_PI / 4 * 3)
        {
            Flag_Qian = 1, Flag_Hou = 0;
            Flag_Left = 0, Flag_Right = 0;
            // printf("前进\r\n");
        }
        else if (a > -Math_PI / 4 * 3 && a <= -Math_PI / 4)
        {
            Flag_Qian = 0, Flag_Hou = 1;
            Flag_Left = 0, Flag_Right = 0;
            // printf("后退\r\n");
        }
        else if (a > -Math_PI / 4 && a <= Math_PI / 4)
        {
            Flag_Left = 0, Flag_Right = 1;
            Flag_Qian = 0, Flag_Hou = 0;
            // printf("右转\r\n");
        }
        else
        {
            Flag_Left = 1, Flag_Right = 0;
            Flag_Qian = 0, Flag_Hou = 0;

            // printf("左转\r\n");
        }
    }
    else if (p < 200)

    {
        Flag_Left = 0, Flag_Right = 0;
        Flag_Qian = 0, Flag_Hou = 0;

        // printf("停止\r\n");
    }


}


/**************************************************************************

入口参数：int
返回  值：unsigned int
**************************************************************************/
int myabs(int a)
{
    int temp;
    if (a < 0)
        temp = -a;
    else
        temp = a;
    return temp;
}


/**************************************************************************
函数功能：对控制输出的PWM线性化,便于给系统寄存器赋值
入口参数：PWM
返回  值：线性化后的PWM，小值变大，大值变小，其结果是moto值相当于转速，转速越小，CCR值越大, 本工程使用的F411芯片，TIM1连到APB2上，速度是100MHz
**************************************************************************/
u16 Linear_Conversion(int moto)
{
    u32 temp;
    u16 Linear_Moto;
    if (moto == 0)
    {
        return 65535; // 或一个极大值，让频率≈0
    }

    temp = 100000000 / (99 + 1) / myabs(moto) / 2; //timer oc toggle 两个周期才能输出一个完整的脉冲，驱动步进电机的一个微步
    if (temp > 65535)
        Linear_Moto = 65535;
    // if(temp > 4000) Linear_Moto = 4000;//计时器的频率为100万每秒，1ms只能读数到1000，5ms只能读数到5000，超过5000的值则在一个控制周期内还未读完，这里ccr值又变大,可能导致永远也不跳转
    else
        Linear_Moto = (u16)temp;
    //    if (temp < 75)
    //        Linear_Moto = 75; // 最小脉冲个数值60个，比这更低马达功率的扭矩不够，转不起来
    //    else
    //        Linear_Moto = (u16)temp;
    static int print_count = 0;
    print_count++;
    if (print_count == 50)
    {
        print_count = 0;

        //printf("Linear_Moto is %d \r\n", Linear_Moto);
    }
    return Linear_Moto; // 计算结果会用1000000/Linear_Moto得到每秒能发多少个脉冲，然后/16/200,就是多少n/s，如果最大转速为
}


/* ================= 主控制循环 ================= */
void ControlLoopPackage()
{
    // 性能测试引脚翻转

// 性能测试引脚翻转
    HAL_GPIO_TogglePin(GPIOC, GPIO_PIN_13); 

    static int last_Moto1 = 0, last_Moto2 = 0;
    last_Moto1 = Moto1;
    last_Moto2 = Moto2;

    // 1. 读取传感器数据 (非阻塞)
    // 假设 mpu_dmp_get_data 返回 0 表示成功，非 0 表示无新数据
    // 即使无新数据，pitch/gyro 全局变量应保持上一次的值，保证循环不卡顿
    mpu6050_data_flag = mpu_dmp_get_data(&pitch, &roll, &yaw);
    
    // 如果传感器完全失效，可根据 flag 处理，这里暂且认为全局变量已更新或保持旧值

    // 2. 读取遥控器
    Get_Elrs(); 

    // 3. 按键启停
    if (click() == 1) {
        Flag_Stop = 0;
        fall_down = 0; // 重置跌倒标志
        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_3, GPIO_PIN_RESET); // 熄灭报警灯
    }

    // 4. 跌倒检测
    if (fabsf(pitch) > 55.0f || fabsf(roll) > 55.0f) {
        fall_down = 1;
        Flag_Stop = 1;
    }

    if (Flag_Stop == 0 && !fall_down)
    {
        // --- A. 自动校准中值 ---
        Auto_Calibrate_Zhongzhi();

        // --- B. 执行核心 PID 算法 ---
        int balance_pwm = 0;
        int turn_pwm = 0;
        
        Balance_PID_Calculate(&balance_pwm, &turn_pwm);

        // --- C. 合成电机输出 ---
        // 左轮 = 直立 + 速度 (已包含在直立逻辑的目标角度偏移中或直接叠加) + 转向
        // 注意：参考项目的速度环输出是角度偏移量，这里为了兼容你的架构，
        // 我们在 Balance_PID_Calculate 内部已经处理了速度环对直立的影响，
        // 或者你可以选择简单的叠加方式：Moto = Balance + Velocity + Turn
        // 下面采用经典的叠加方式 (需根据实际调试正负号)
        
        // 由于上面的 Balance_PID_Calculate 设计为串级，它直接输出最终的直立分量
        // 如果需要单独的速度分量叠加，请调整函数返回值。
        // 这里假设 balance_pwm 已经是 (直立环 + 速度环修正) 的结果
        
        Moto1 = balance_pwm + turn_pwm;
        Moto2 = balance_pwm - turn_pwm;

        // --- D. 限幅与死区处理 (适配你的 Linear_Conversion) ---
        // 限幅
        if (Moto1 > MAX_MOTO) Moto1 = MAX_MOTO;
        if (Moto1 < -MAX_MOTO) Moto1 = -MAX_MOTO;
        if (Moto2 > MAX_MOTO) Moto2 = MAX_MOTO;
        if (Moto2 < -MAX_MOTO) Moto2 = -MAX_MOTO;

        // 死区 (防止 moto=0 导致 Linear_Conversion 除零)
        if (Moto1 != 0 && myabs(Moto1) < MOTO_DEADZONE) 
            Moto1 = (Moto1 > 0) ? MOTO_DEADZONE : -MOTO_DEADZONE;
        if (Moto2 != 0 && myabs(Moto2) < MOTO_DEADZONE) 
            Moto2 = (Moto2 > 0) ? MOTO_DEADZONE : -MOTO_DEADZONE;

        // --- E. 安全检查与输出 ---
        if (Turn_Off(roll) == 0) // 正常
        {
            ST = 0; // 电机使能
            Set_Pwm(Moto1, Moto2);
        }
        else // 异常 (如解锁未就绪)
        {
            ST = 1;
            Moto1 = 0; Moto2 = 0;
            Set_Pwm(Moto1, Moto2);
            Flag_Stop = 1;
            HAL_GPIO_WritePin(GPIOB, GPIO_PIN_3, GPIO_PIN_SET);
        }
    }
    else
    {
        // 停止或跌倒处理
        Moto1 = 0; Moto2 = 0;
        Set_Pwm(Moto1, Moto2);
        ST = 1; // 失能
        
        // 重置 PID 状态
        vel_integral = 0; smooth_target_vel = 0; estimated_speed = 0; last_vel_output = 0;
        turn_integral = 0; smooth_turn_output = 0;
        ang_e_k_1 = 0; ang_e_k_2 = 0;
        pal_e_k_1 = 0; pal_e_k_2 = 0;
        
        // 重置校准状态
        AutoZero_SumAngle = 0; AutoZero_Count = 0; AutoZero_InProgress = 0; start_delay = 0;
        Flag_Zhongzhi = 0; 
        
        if(fall_down) {
             ZeroRequirementMean = 1;
             ZeroRequirementSpeedPid = 1;
             HAL_GPIO_WritePin(GPIOB, GPIO_PIN_3, GPIO_PIN_SET); // 报警
        }
    }

    // 性能测试引脚翻转
    HAL_GPIO_TogglePin(GPIOC, GPIO_PIN_13); 
}

/**
 * @brief 核心 PID 计算函数 (移植参考项目算法)
 */
void Balance_PID_Calculate(int *p_balance_pwm, int *p_turn_pwm)
{
    float dt = CONTROL_DT;
    float angle_err, gyro_err, vel_err;
    float kp, ki, kd;
    float output_ang, output_pal, output_vel, output_turn;
    
    // ================= 1. 速度环 (Velocity Loop) =================
    // 目标：通过改变“目标角度”来控制速度
    float raw_target_vel = 0;
    if (Flag_Qian) raw_target_vel = (float)Target_Velocity;
    else if (Flag_Hou) raw_target_vel = -(float)Target_Velocity;
    
    // 软启动 (Ramp)
    if (raw_target_vel > smooth_target_vel) {
        smooth_target_vel += VEL_RAMP_STEP;
        if (smooth_target_vel > raw_target_vel) smooth_target_vel = raw_target_vel;
    } else if (raw_target_vel < smooth_target_vel) {
        smooth_target_vel -= VEL_RAMP_STEP;
        if (smooth_target_vel < raw_target_vel) smooth_target_vel = raw_target_vel;
    }
    
    // 死区停车
    if (fabsf(smooth_target_vel) < 50.0f && fabsf(raw_target_vel) < 50.0f) {
        smooth_target_vel = 0;
        vel_integral = 0; estimated_speed = 0; last_vel_output = 0;
        output_vel = 0;
    } else {
        // 估算速度 (开环模型：用上一轮输出模拟)
        estimated_speed = estimated_speed * 0.92f + last_vel_output * 0.08f;
        
        vel_err = smooth_target_vel - estimated_speed;
        
        // PI 计算 (参考项目逻辑简化版)
        kp = KP_VEL;
        ki = KP_VEL * dt / TI_VEL;
        // output = P*(e_k - e_k_1) + I*e_k + D*...
        // 这里简化为经典 PI: P*e + I*integral
        vel_integral += vel_err;
        // 积分限幅
        if (vel_integral > 5000) vel_integral = 5000;
        if (vel_integral < -5000) vel_integral = -5000;
        
        output_vel = (vel_err * kp) + (vel_integral * ki);
        last_vel_output = output_vel;
    }
    
    // 速度环的输出作为“角度偏移量”叠加到平衡中值上
    // 系数 0.05 需调试：速度误差越大，车身需要倾斜越多来加速
    float target_angle = Zhongzhi + (output_vel * 0.05f); 

    // ================= 2. 直立角度环 (Angle Loop) =================
    // 输入：目标角度 (来自速度环) vs 实际角度 (pitch 或 roll，取决于安装)
    // 假设你的车是前后平衡，用 pitch；如果是左右，用 roll。原代码用 -roll，这里保持一致
    float current_angle = -roll; 
    angle_err = target_angle - current_angle;
    
    kp = KP_ANG;
    ki = KP_ANG * dt / TI_ANG;
    kd = KP_ANG * TD_ANG / dt;
    
    // 位置式 PID
    output_ang = kp * (angle_err - ang_e_k_1) + ki * angle_err + kd * (angle_err - 2*ang_e_k_1 + ang_e_k_2);
    
    ang_e_k_2 = ang_e_k_1;
    ang_e_k_1 = angle_err;
    
    // 角度环的输出作为“目标角速度”传给内环
    int target_gyro = (int)output_ang;

    // ================= 3. 角速度环 (Gyro/Pal Loop) =================
    // 输入：目标角速度 vs 实际角速度 (gyroy 或 gyrox)
    // 原代码用 gyrox 对应 roll，这里保持一致
    float current_gyro = (float)gyrox; 
    gyro_err = (float)target_gyro - current_gyro;
    
    kp = KP_PAL;
    ki = KP_PAL * dt / TI_PAL;
    kd = KP_PAL * TD_PAL / dt;
    
    output_pal = kp * (gyro_err - pal_e_k_1) + ki * gyro_err + kd * (gyro_err - 2*pal_e_k_1 + pal_e_k_2);
    
    pal_e_k_2 = pal_e_k_1;
    pal_e_k_1 = gyro_err;
    
    // 最终直立 PWM (加一个缩放系数)
    int balance_out = (int)(output_pal * 0.8f);

    // ================= 4. 转向环 (Turn Loop) =================
    float raw_turn_target = 0;
    if (Flag_Left) raw_turn_target = -TURN_GAIN;
    else if (Flag_Right) raw_turn_target = TURN_GAIN;
    
    // 一阶低通滤波
    smooth_turn_output = smooth_turn_output * (1.0f - TURN_FILTER_COEF) + raw_turn_target * TURN_FILTER_COEF;
    
    // 死区归零
    if (raw_turn_target == 0 && fabsf(smooth_turn_output) < 5.0f) {
        smooth_turn_output = 0;
    }
    
    // 可选：加入陀螺仪反馈 (gyroz) 增加稳定性
    // float turn_err = smooth_turn_output - (float)gyroz;
    // 这里简化为直接输出滤波后的值，如需 PID 可展开
    int turn_out = (int)smooth_turn_output;

    // 输出结果
    *p_balance_pwm = balance_out;
    *p_turn_pwm = turn_out;
}

/**************************************************************************
函数功能：自适应中值 (改进版：允许小幅摆动中校准)
入口参数：无
返回  值：无
**************************************************************************/
void Get_Zhongzhi(void)
{
    // 如果已经校准过，直接退出 (除非你想做实时动态校准，否则校准一次即可)
    if (Flag_Zhongzhi == 1) return;

    // 安全检测：如果车快倒了 (角度过大)，重置采样
    if (fabs(roll) > 15.0f) 
    {
        AutoZero_SumAngle = 0;
        AutoZero_Count = 0;
        AutoZero_InProgress = 0;
        return;
    }

    // 启动条件：
    // 1. 电机已使能 (Flag_Stop == 0)
    // 2. 角度在合理范围内 (-10度 到 10度)
    // 3. 已经运行了一小段时间 (避免上电瞬间的瞬态干扰)，比如进入控制循环 100 次后
    static int start_delay = 0;
    if (Flag_Stop == 0) start_delay++;
    
    if (start_delay > 100 && fabs(roll) < 10.0f) 
    {
        AutoZero_InProgress = 1;
    }

    // 执行采样
    if (AutoZero_InProgress == 1)
    {
        AutoZero_SumAngle += (-roll); // 累加当前角度 (注意符号，你的 balance 函数用的是 -roll)
        AutoZero_Count++;

        // 采样足够次数 (假设 1kHz 中断，采样 2000 次 = 2 秒)
        // 2 秒内车会左右晃动几个周期，平均值即为机械中值
        if (AutoZero_Count >= 1000) 
        {
            // 计算平均值
            float new_zhongzhi = AutoZero_SumAngle / AutoZero_Count;
            
            // 【关键】平滑过渡：不要直接赋值，防止突变导致摔倒
            // 如果新旧值差别不大，直接更新；如果差别大，分步更新或直接更新但配合低速启动
            Zhongzhi = new_zhongzhi;
            Flag_Zhongzhi = 1; // 标记校准完成
            
            // 提示用户
            HAL_GPIO_WritePin(GPIOC, GPIO_PIN_14,GPIO_PIN_SET); // 亮红灯表示中值已经找到
            printf("Zhongzhi Auto-Calibrated: %f (Samples: %d)\r\n", Zhongzhi, AutoZero_Count);
            
            // 重置采样状态
            AutoZero_SumAngle = 0;
            AutoZero_Count = 0;
            AutoZero_InProgress = 0;
        }
    }
}

int map(int val, int I_Min, int I_Max, int O_Min, int O_Max)
{
  return (val - I_Min) * (O_Max - O_Min) / (I_Max - I_Min) + O_Min;
}
